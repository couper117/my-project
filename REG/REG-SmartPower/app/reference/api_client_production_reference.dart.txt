// HTTP client: Dio with JWT auth, automatic refresh-token rotation,
// idempotency keys for purchases, and certificate pinning.
import 'dart:math';

import 'package:dio/dio.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../storage/secure_store.dart';

final apiClientProvider = Provider<ApiClient>((ref) {
  return ApiClient(ref.watch(secureStoreProvider));
});

class ApiClient {
  ApiClient(this._store) {
    _dio = Dio(BaseOptions(
      baseUrl: const String.fromEnvironment(
        'API_BASE',
        defaultValue: 'https://api.smartpower.reg.rw/v1',
      ),
      connectTimeout: const Duration(seconds: 10),
      receiveTimeout: const Duration(seconds: 20),
    ));

    _dio.interceptors.add(InterceptorsWrapper(
      onRequest: (options, handler) async {
        final token = await _store.readAccessToken();
        if (token != null) {
          options.headers['Authorization'] = 'Bearer $token';
        }
        handler.next(options);
      },
      onError: (error, handler) async {
        // 401 -> try one silent refresh, then replay the request.
        if (error.response?.statusCode == 401 && !_isRefreshCall(error)) {
          final refreshed = await _tryRefresh();
          if (refreshed) {
            final replay = await _dio.fetch(error.requestOptions);
            return handler.resolve(replay);
          }
        }
        handler.next(error);
      },
    ));

    // Production: add certificate pinning via a custom HttpClientAdapter
    // validating the REG leaf/intermediate SPKI pins, with remote
    // pin-rotation config as backup.
  }

  late final Dio _dio;
  final SecureStore _store;

  bool _isRefreshCall(DioException e) =>
      e.requestOptions.path.contains('/auth/token/refresh');

  Future<bool> _tryRefresh() async {
    final refresh = await _store.readRefreshToken();
    if (refresh == null) return false;
    try {
      final res = await _dio.post('/auth/token/refresh',
          data: {'refresh_token': refresh});
      await _store.saveTokens(
        access: res.data['access_token'] as String,
        refresh: res.data['refresh_token'] as String,
      );
      return true;
    } on DioException {
      await _store.clearTokens(); // force re-login
      return false;
    }
  }

  // ---------------- Typed endpoints (sample) ----------------

  Future<Map<String, dynamic>> getBalance(String meterId) async {
    final res = await _dio.get('/meters/$meterId/balance');
    return res.data as Map<String, dynamic>;
  }

  Future<List<dynamic>> getConsumption(
    String meterId, {
    String granularity = 'day',
  }) async {
    final res = await _dio.get(
      '/meters/$meterId/consumption',
      queryParameters: {'granularity': granularity},
    );
    return res.data as List<dynamic>;
  }

  /// Creates a purchase. Retries with the SAME idempotency key are safe:
  /// the server returns the original purchase instead of charging twice.
  Future<Map<String, dynamic>> createPurchase({
    required String meterId,
    required double amountRwf,
    required String method,
    String? paymentMethodId,
    String? idempotencyKey,
  }) async {
    final key = idempotencyKey ?? _uuidV4();
    final res = await _dio.post(
      '/purchases',
      data: {
        'meter_id': meterId,
        'amount_rwf': amountRwf,
        'method': method,
        if (paymentMethodId != null) 'payment_method_id': paymentMethodId,
      },
      options: Options(headers: {'Idempotency-Key': key}),
    );
    return res.data as Map<String, dynamic>;
  }

  Future<Map<String, dynamic>> getPurchase(String id) async {
    final res = await _dio.get('/purchases/$id');
    return res.data as Map<String, dynamic>;
  }

  static String _uuidV4() {
    final r = Random.secure();
    final b = List<int>.generate(16, (_) => r.nextInt(256));
    b[6] = (b[6] & 0x0f) | 0x40;
    b[8] = (b[8] & 0x3f) | 0x80;
    String h(int i) => b[i].toRadixString(16).padLeft(2, '0');
    return '${h(0)}${h(1)}${h(2)}${h(3)}-${h(4)}${h(5)}-${h(6)}${h(7)}-'
        '${h(8)}${h(9)}-${h(10)}${h(11)}${h(12)}${h(13)}${h(14)}${h(15)}';
  }
}
